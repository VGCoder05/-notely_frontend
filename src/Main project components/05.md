```javascript
    auth: authReducer,
```

### **Component 41: EditorView**

**`src/pages/EditorView/EditorView.logic.js`**
```javascript
import { useEffect } from 'react';
import { useParams, useNavigate } from 'react-router-dom';
import { useSelector, useDispatch } from 'react-redux';
import { setBreadcrumb } from '../../store/slices/uiSlice';
import { setCurrentNote, addNote } from '../../store/slices/notesSlice';

export const useEditorView = () => {
  const { noteId } = useParams();
  const navigate = useNavigate();
  const dispatch = useDispatch();

  const currentNote = useSelector((state) => state.notes.currentNote);
  const notes = useSelector((state) => state.notes.notes);

  useEffect(() => {
    if (noteId === 'new') {
      // Create new note
      const newNote = {
        id: Date.now().toString(),
        title: '',
        content: '',
        tags: [],
        isFavourite: false,
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString(),
      };
      dispatch(addNote(newNote));
      dispatch(setCurrentNote(newNote));
      dispatch(setBreadcrumb('New note'));
      
      // Replace URL with actual note ID
      navigate(`/note/${newNote.id}`, { replace: true });
    } else {
      // Load existing note
      const note = notes.find((n) => n.id === noteId);
      
      if (note) {
        dispatch(setCurrentNote(note));
        dispatch(setBreadcrumb(note.title || 'Untitled'));
      } else {
        // Note not found
        navigate('/dashboard', { replace: true });
      }
    }
  }, [noteId, notes, dispatch, navigate]);

  // Update breadcrumb when note title changes
  useEffect(() => {
    if (currentNote) {
      dispatch(setBreadcrumb(currentNote.title || 'Untitled'));
    }
  }, [currentNote?.title, dispatch]);

  return {
    note: currentNote,
    noteId,
  };
};
```

```javascript
    handlePermanentDelete,
    handleEmptyTrash,
  };
```
