### **Hook 1: useAutoSave** *(Already in NoteEditor.logic.js but extracted)*

**`src/hooks/useAutoSave.js`**
```javascript
import { useState, useEffect, useCallback, useRef } from 'react';

/**
 * Auto-save hook with debouncing
 * @param {Function} saveFn - Function to call when saving
 * @param {any} data - Data to save
 * @param {number} delay - Debounce delay in ms
 * @returns {Object} - { saveStatus, forceSave }
 */
export const useAutoSave = (saveFn, data, delay = 1000) => {
  const [saveStatus, setSaveStatus] = useState('saved'); // 'saved' | 'saving' | 'unsaved'
  const timeoutRef = useRef(null);
  const isMountedRef = useRef(false);
  const previousDataRef = useRef(data);

  const performSave = useCallback(async () => {
    if (!saveFn) return;

    setSaveStatus('saving');

    try {
      await saveFn(data);
      setSaveStatus('saved');
    } catch (error) {
      setSaveStatus('unsaved');
      console.error('Auto-save failed:', error);
    }
  }, [saveFn, data]);

  const forceSave = useCallback(() => {
    if (timeoutRef.current) {
      clearTimeout(timeoutRef.current);
    }
    performSave();
  }, [performSave]);

  useEffect(() => {
    // Don't auto-save on initial mount
    if (!isMountedRef.current) {
      isMountedRef.current = true;
      previousDataRef.current = data;
      return;
    }

    // Don't save if data hasn't changed
    if (JSON.stringify(data) === JSON.stringify(previousDataRef.current)) {
      return;
    }

    previousDataRef.current = data;
    setSaveStatus('unsaved');

    // Clear existing timeout
    if (timeoutRef.current) {
      clearTimeout(timeoutRef.current);
    }

    // Set new timeout for auto-save
    timeoutRef.current = setTimeout(() => {
      performSave();
    }, delay);

    return () => {
      if (timeoutRef.current) {
        clearTimeout(timeoutRef.current);
      }
    };
  }, [data, delay, performSave]);

  return {
    saveStatus,
    forceSave,
  };
};
```

### **Hook 2: useNotes** *(Complete Notes Operations)*

**`src/hooks/useNotes.js`**

```javascript
import { useCallback, useMemo } from 'react';
import { useDispatch, useSelector } from 'react-redux';
import {
  setNotes,
  addNote,
  updateNote,
  deleteNote,
  toggleFavourite,
  setCurrentNote,
  setLoading,
  setError,
  searchNotes,
  filterByTag,
  sortNotes,
} from '../store/slices/notesSlice';
import { toast } from 'react-toastify';

export const useNotes = () => {
  const dispatch = useDispatch();

  const notes = useSelector((state) => state.notes.notes);
  const currentNote = useSelector((state) => state.notes.currentNote);
  const loading = useSelector((state) => state.notes.loading);
  const error = useSelector((state) => state.notes.error);
  const searchQuery = useSelector((state) => state.notes.searchQuery);
  const selectedTags = useSelector((state) => state.notes.selectedTags);
  const sortBy = useSelector((state) => state.notes.sortBy);

  // Filtered and sorted notes
  const filteredNotes = useMemo(() => {
    let result = [...notes];

    // Apply search filter
    if (searchQuery) {
      const query = searchQuery.toLowerCase();
      result = result.filter((note) => {
        const titleMatch = note.title?.toLowerCase().includes(query);
        const contentMatch = note.content
          ?.replace(/<[^>]*>/g, '')
          .toLowerCase()
          .includes(query);
        const tagMatch = note.tags?.some((tag) =>
          tag.toLowerCase().includes(query)
        );
        return titleMatch || contentMatch || tagMatch;
      });
    }

    // Apply tag filter
    if (selectedTags.length > 0) {
      result = result.filter((note) =>
        selectedTags.some((tag) => note.tags?.includes(tag))
      );
    }

    // Apply sorting
    switch (sortBy) {
      case 'updated':
        result.sort(
          (a, b) => new Date(b.updatedAt) - new Date(a.updatedAt)
        );
        break;
      case 'created':
        result.sort(
          (a, b) => new Date(b.createdAt) - new Date(a.createdAt)
        );
        break;
      case 'title':
        result.sort((a, b) => {
          const titleA = a.title?.toLowerCase() || '';
          const titleB = b.title?.toLowerCase() || '';
          return titleA.localeCompare(titleB);
        });
        break;
      default:
        break;
    }

    return result;
  }, [notes, searchQuery, selectedTags, sortBy]);

  // Favourite notes
  const favouriteNotes = useMemo(() => {
    return notes.filter((note) => note.isFavourite && !note.isDeleted);
  }, [notes]);

  // Trashed notes
  const trashedNotes = useMemo(() => {
    return notes.filter((note) => note.isDeleted);
  }, [notes]);

  // All tags
  const allTags = useMemo(() => {
    const tagSet = new Set();
    notes.forEach((note) => {
      note.tags?.forEach((tag) => tagSet.add(tag));
    });
    return Array.from(tagSet).sort();
  }, [notes]);

  // Create note
  const createNote = useCallback(
    (noteData = {}) => {
      const newNote = {
        id: Date.now().toString(),
        title: '',
        content: '',
        tags: [],
        isFavourite: false,
        isDeleted: false,
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString(),
        ...noteData,
      };

      dispatch(addNote(newNote));
      dispatch(setCurrentNote(newNote));
      toast.success('Note created');
      return newNote;
    },
    [dispatch]
  );

  // Update note
  const saveNote = useCallback(
    (noteId, updates) => {
      const updatedNote = {
        ...notes.find((n) => n.id === noteId),
        ...updates,
        updatedAt: new Date().toISOString(),
      };

      dispatch(updateNote(updatedNote));
      return updatedNote;
    },
    [dispatch, notes]
  );

  // Delete note (move to trash)
  const trashNote = useCallback(
    (noteId) => {
      const note = notes.find((n) => n.id === noteId);
      if (note) {
        dispatch(
          updateNote({
            ...note,
            isDeleted: true,
            deletedAt: new Date().toISOString(),
          })
        );
        toast.success('Note moved to trash');
      }
    },
    [dispatch, notes]
  );

  // Restore note from trash
  const restoreNote = useCallback(
    (noteId) => {
      const note = notes.find((n) => n.id === noteId);
      if (note) {
        dispatch(
          updateNote({
            ...note,
            isDeleted: false,
            deletedAt: null,
          })
        );
        toast.success('Note restored');
      }
    },
    [dispatch, notes]
  );

  // Permanently delete note
  const permanentlyDeleteNote = useCallback(
    (noteId) => {
      dispatch(deleteNote(noteId));
      toast.success('Note permanently deleted');
    },
    [dispatch]
  );

  // Toggle favourite
  const toggleNoteFavourite = useCallback(
    (noteId) => {
      dispatch(toggleFavourite(noteId));
      const note = notes.find((n) => n.id === noteId);
      toast.success(
        note?.isFavourite ? 'Removed from favourites' : 'Added to favourites'
      );
    },
    [dispatch, notes]
  );

  // Search notes
  const search = useCallback(
    (query) => {
      dispatch(searchNotes(query));
    },
    [dispatch]
  );

  // Filter by tags
  const filterTags = useCallback(
    (tags) => {
      dispatch(filterByTag(tags));
    },
    [dispatch]
  );

  // Sort notes
  const sort = useCallback(
    (sortOption) => {
      dispatch(sortNotes(sortOption));
    },
    [dispatch]
  );

  // Load notes (simulate API call)
  const loadNotes = useCallback(async () => {
    dispatch(setLoading(true));
    try {
      // Simulate API delay
      await new Promise((resolve) => setTimeout(resolve, 500));

      // Load from localStorage
      const savedNotes = JSON.parse(localStorage.getItem('notes') || '[]');
      dispatch(setNotes(savedNotes));
    } catch (err) {
      dispatch(setError(err.message));
      toast.error('Failed to load notes');
    } finally {
      dispatch(setLoading(false));
    }
  }, [dispatch]);

  // Save notes to localStorage
  const persistNotes = useCallback(() => {
    try {
      localStorage.setItem('notes', JSON.stringify(notes));
    } catch (err) {
      console.error('Failed to persist notes:', err);
    }
  }, [notes]);

  return {
    // State
    notes: filteredNotes,
    allNotes: notes,
    currentNote,
    favouriteNotes,
    trashedNotes,
    allTags,
    loading,
    error,
    searchQuery,
    selectedTags,
    sortBy,

    // Actions
    createNote,
    saveNote,
    trashNote,
    restoreNote,
    permanentlyDeleteNote,
    toggleNoteFavourite,
    search,
    filterTags,
    sort,
    loadNotes,
    persistNotes,
    setCurrentNote: (note) => dispatch(setCurrentNote(note)),
  };
};
```

### **Hook 3: useLocalStorage**

**`src/hooks/useLocalStorage.js`**
```javascript
import { useState, useEffect, useCallback } from 'react';

/**
 * Sync state with localStorage
 * @param {string} key - localStorage key
 * @param {any} initialValue - Initial value if key doesn't exist
 * @returns {[any, Function, Function]} - [value, setValue, removeValue]
 */
export const useLocalStorage = (key, initialValue) => {
  // Get from local storage then parse stored json or return initialValue
  const readValue = useCallback(() => {
    if (typeof window === 'undefined') {
      return initialValue;
    }

    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catch (error) {
      console.warn(`Error reading localStorage key "${key}":`, error);
      return initialValue;
    }
  }, [initialValue, key]);

  const [storedValue, setStoredValue] = useState(readValue);

  // Return a wrapped version of useState's setter function that persists the new value to localStorage
  const setValue = useCallback(
    (value) => {
      if (typeof window === 'undefined') {
        console.warn(
          `Tried setting localStorage key "${key}" even though environment is not a client`
        );
      }

      try {
        // Allow value to be a function so we have the same API as useState
        const newValue = value instanceof Function ? value(storedValue) : value;

        // Save to local storage
        window.localStorage.setItem(key, JSON.stringify(newValue));

        // Save state
        setStoredValue(newValue);

        // Dispatch custom event so every useLocalStorage hook is notified
        window.dispatchEvent(new Event('local-storage'));
      } catch (error) {
        console.warn(`Error setting localStorage key "${key}":`, error);
      }
    },
    [key, storedValue]
  );

  const removeValue = useCallback(() => {
    try {
      window.localStorage.removeItem(key);
      setStoredValue(initialValue);
      window.dispatchEvent(new Event('local-storage'));
    } catch (error) {
      console.warn(`Error removing localStorage key "${key}":`, error);
    }
  }, [key, initialValue]);

  // Listen for changes in other tabs/windows
  useEffect(() => {
    const handleStorageChange = () => {
      setStoredValue(readValue());
    };

    window.addEventListener('storage', handleStorageChange);
    window.addEventListener('local-storage', handleStorageChange);

    return () => {
      window.removeEventListener('storage', handleStorageChange);
      window.removeEventListener('local-storage', handleStorageChange);
    };
  }, [readValue]);

  return [storedValue, setValue, removeValue];
};
```

**`src/hooks/useKeyboardShortcut.js`**

```javascript
import { useEffect, useCallback } from 'react';

/**
 * Register keyboard shortcuts
 * @param {Object} shortcuts - { 'ctrl+k': handler, 'n': handler, etc. }
 * @param {Array} deps - Dependencies array
 */
export const useKeyboardShortcut = (shortcuts, deps = []) => {
  const handleKeyDown = useCallback(
    (event) => {
      // Don't trigger if user is typing in an input/textarea
      if (['INPUT', 'TEXTAREA'].includes(event.target.tagName)) {
        return;
      }

      const key = event.key.toLowerCase();
      const ctrl = event.ctrlKey || event.metaKey;
      const shift = event.shiftKey;
      const alt = event.altKey;

      // Build shortcut string
      let shortcut = '';
      if (ctrl) shortcut += 'ctrl+';
      if (shift) shortcut += 'shift+';
      if (alt) shortcut += 'alt+';
      shortcut += key;

      // Also check for just the key
      const handlers = [shortcuts[shortcut], shortcuts[key]].filter(Boolean);

      if (handlers.length > 0) {
        event.preventDefault();
        handlers.forEach((handler) => handler(event));
      }
    },
    [shortcuts]
  );

  useEffect(() => {
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [handleKeyDown, ...deps]); // eslint-disable-line react-hooks/exhaustive-deps
};
```

```javascript
import { useState, useEffect } from 'react';

/**
 * Track online/offline status
 * @returns {boolean} - Is online
 */
export const useOnlineStatus = () => {
  const [isOnline, setIsOnline] = useState(
    typeof navigator !== 'undefined' ? navigator.onLine : true
  );

  useEffect(() => {
    const handleOnline = () => setIsOnline(true);
    const handleOffline = () => setIsOnline(false);

    window.addEventListener('online', handleOnline);
    window.addEventListener('offline', handleOffline);

    return () => {
      window.removeEventListener('online', handleOnline);
      window.removeEventListener('offline', handleOffline);
    };
  }, []);

  return isOnline;
};
```

```javascript
import { useState, useCallback } from 'react';

/**
 * Confirmation dialog hook
 * @returns {Object} - { confirm, ConfirmDialog }
 */
export const useConfirm = () => {
  const [isOpen, setIsOpen] = useState(false);
  const [config, setConfig] = useState({
    title: 'Confirm',
    message: 'Are you sure?',
    confirmText: 'Confirm',
    cancelText: 'Cancel',
  });
  const [resolveRef, setResolveRef] = useState(null);

  const confirm = useCallback((options = {}) => {
    return new Promise((resolve) => {
      setConfig({
        title: options.title || 'Confirm',
        message: options.message || 'Are you sure?',
        confirmText: options.confirmText || 'Confirm',
        cancelText: options.cancelText || 'Cancel',
        variant: options.variant || 'primary', // 'primary' | 'danger'
      });
      setResolveRef(() => resolve);
      setIsOpen(true);
    });
  }, []);

  const handleConfirm = useCallback(() => {
    resolveRef?.(true);
    setIsOpen(false);
  }, [resolveRef]);

  const handleCancel = useCallback(() => {
    resolveRef?.(false);
    setIsOpen(false);
  }, [resolveRef]);

  const ConfirmDialog = useCallback(() => {
    if (!isOpen) return null;

    return (
      <div
        onClick={handleCancel}
        className="fixed inset-0 bg-[var(--color-overlay-dark)] z-[var(--z-index-modal)] grid place-items-center p-4"
      >
        <div
          onClick={(e) => e.stopPropagation()}
          className="w-[min(100%,400px)] bg-white rounded-[14px] shadow-[var(--shadow-lg)] overflow-hidden"
        >
          <div className="p-5">
            <h3 className="text-[18px] font-bold m-0 mb-2">{config.title}</h3>
            <p className="text-[var(--color-text-secondary)] m-0">
              {config.message}
            </p>
          </div>
          <div className="flex gap-2 p-4 border-t border-[var(--color-border-primary)]">
            <button
              onClick={handleCancel}
              className="flex-1 min-h-[40px] px-4 rounded-[9px] border border-[var(--color-border-primary)] bg-white font-semibold"
            >
              {config.cancelText}
            </button>
            <button
              onClick={handleConfirm}
              className={`flex-1 min-h-[40px] px-4 rounded-[9px] border font-semibold text-white ${

                  : 'bg-[var(--color-primary-500)] border-[var(--color-primary-500)]'
              }`}
            >
              {config.confirmText}
            </button>
          </div>
        </div>
      </div>
    );
  }, [isOpen, config, handleConfirm, handleCancel]);

  return { confirm, ConfirmDialog };
};
```

```javascript
import { createSlice, createAsyncThunk } from '@reduxjs/toolkit';

// Async thunk for loading notes (simulated API)
export const fetchNotes = createAsyncThunk(

// Async thunk for saving notes

const initialState = {
  notes: [],
  recentNotes: [],
  currentNote: null,
  
  // Loading states
  loading: false,
  saving: false,
  error: null,
  
  // Filtering & search
  searchQuery: '',
  selectedTags: [],
  sortBy: 'updated', // 'updated' | 'created' | 'title'
  viewMode: 'grid', // 'grid' | 'list'
  
  // Statistics
  stats: {
    total: 0,
    favourites: 0,
    trash: 0,
  },
};

const notesSlice = createSlice({
  name: 'notes',
  initialState,
  reducers: {
    // CRUD Operations
    setNotes: (state, action) => {
      state.notes = action.payload;
      updateStats(state);
      updateRecentNotes(state);
    },
    
    addNote: (state, action) => {
      state.notes.unshift(action.payload);
      updateStats(state);
      updateRecentNotes(state);
      persistToStorage(state.notes);
    },
    
    updateNote: (state, action) => {
      const index = state.notes.findIndex((note) => note.id === action.payload.id);
      if (index !== -1) {
        state.notes[index] = {
          ...state.notes[index],
          ...action.payload,
          updatedAt: new Date().toISOString(),
        };
        
        // Update current note if it's the one being edited
        if (state.currentNote?.id === action.payload.id) {
          state.currentNote = state.notes[index];
        }
        
        updateStats(state);
        updateRecentNotes(state);
        persistToStorage(state.notes);
      }
    },
    
    deleteNote: (state, action) => {
      state.notes = state.notes.filter((note) => note.id !== action.payload);
      state.recentNotes = state.recentNotes.filter((note) => note.id !== action.payload);
      
      if (state.currentNote?.id === action.payload) {
        state.currentNote = null;
      }
      
      updateStats(state);
      persistToStorage(state.notes);
    },
    
    deleteMultipleNotes: (state, action) => {
      const idsToDelete = new Set(action.payload);
      state.notes = state.notes.filter((note) => !idsToDelete.has(note.id));
      state.recentNotes = state.recentNotes.filter((note) => !idsToDelete.has(note.id));
      
      if (state.currentNote && idsToDelete.has(state.currentNote.id)) {
        state.currentNote = null;
      }
      
      updateStats(state);
      persistToStorage(state.notes);
    },
    
    // Favourite operations
    toggleFavourite: (state, action) => {
      const note = state.notes.find((note) => note.id === action.payload);
      if (note) {
        note.isFavourite = !note.isFavourite;
        note.updatedAt = new Date().toISOString();
        updateStats(state);
        persistToStorage(state.notes);
      }
    },
    
    addToFavourites: (state, action) => {
      const note = state.notes.find((note) => note.id === action.payload);
      if (note) {
        note.isFavourite = true;
        updateStats(state);
        persistToStorage(state.notes);
      }
    },
    
    removeFromFavourites: (state, action) => {
      const note = state.notes.find((note) => note.id === action.payload);
      if (note) {
        note.isFavourite = false;
        updateStats(state);
        persistToStorage(state.notes);
      }
    },
    
    // Trash operations
    moveToTrash: (state, action) => {
      const note = state.notes.find((note) => note.id === action.payload);
      if (note) {
        note.isDeleted = true;
        note.deletedAt = new Date().toISOString();
        updateStats(state);
        persistToStorage(state.notes);
      }
    },
    
    restoreFromTrash: (state, action) => {
      const note = state.notes.find((note) => note.id === action.payload);
      if (note) {
        note.isDeleted = false;
        note.deletedAt = null;
        updateStats(state);
        persistToStorage(state.notes);
      }
    },
    
    emptyTrash: (state) => {
      state.notes = state.notes.filter((note) => !note.isDeleted);
      updateStats(state);
      persistToStorage(state.notes);
    },
    
    // Current note
    setCurrentNote: (state, action) => {
      state.currentNote = action.payload;
    },
    
    clearCurrentNote: (state) => {
      state.currentNote = null;
    },
    
    // Search & Filter
    searchNotes: (state, action) => {
      state.searchQuery = action.payload;
    },
    
    clearSearch: (state) => {
      state.searchQuery = '';
    },
    
    filterByTag: (state, action) => {
      state.selectedTags = action.payload;
    },
    
    addTagFilter: (state, action) => {
      if (!state.selectedTags.includes(action.payload)) {
        state.selectedTags.push(action.payload);
      }
    },
    
    removeTagFilter: (state, action) => {
      state.selectedTags = state.selectedTags.filter((tag) => tag !== action.payload);
    },
    
    clearFilters: (state) => {
      state.searchQuery = '';
      state.selectedTags = [];
    },
    
    // Sorting
    sortNotes: (state, action) => {
      state.sortBy = action.payload;
    },
    
    // View mode
    setViewMode: (state, action) => {
      state.viewMode = action.payload;
    },
    
    // Recent notes
    setRecentNotes: (state, action) => {
      state.recentNotes = action.payload.slice(0, 5);
    },
    
    // Loading states
    setLoading: (state, action) => {
      state.loading = action.payload;
    },
    
    setSaving: (state, action) => {
      state.saving = action.payload;
    },
    
    setError: (state, action) => {
      state.error = action.payload;
    },
    
    clearError: (state) => {
      state.error = null;
    },
    
    // Bulk operations
    duplicateNote: (state, action) => {
      const originalNote = state.notes.find((note) => note.id === action.payload);
      if (originalNote) {
        const duplicatedNote = {
          ...originalNote,
          id: Date.now().toString(),
          title: `${originalNote.title} (Copy)`,
          createdAt: new Date().toISOString(),
          updatedAt: new Date().toISOString(),
          isFavourite: false,
        };
        state.notes.unshift(duplicatedNote);
        updateStats(state);
        persistToStorage(state.notes);
      }
    },
    
    updateNoteTags: (state, action) => {
      const { noteId, tags } = action.payload;
      const note = state.notes.find((n) => n.id === noteId);
      if (note) {
        note.tags = tags;
        note.updatedAt = new Date().toISOString();
        persistToStorage(state.notes);
      }
```

```javascript
// Load persisted state from localStorage
const loadState = () => {
  try {
    const serializedState = localStorage.getItem('notelyState');
    if (serializedState === null) {
      return undefined;
    }
    return JSON.parse(serializedState);
  } catch (err) {
    console.warn('Failed to load state:', err);
    return undefined;
  }

// Save state to localStorage (throttled)
let saveTimeout;
const saveState = (state) => {
  clearTimeout(saveTimeout);
  saveTimeout = setTimeout(() => {
    try {
      const serializedState = JSON.stringify({
        notes: state.notes,
        auth: {
          user: state.auth.user,
          isAuthenticated: state.auth.isAuthenticated,
        },
        settings: state.settings,
        ui: {
          theme: state.ui.theme,
          layout: state.ui.layout,

        },
      });
      localStorage.setItem('notelyState', serializedState);
    } catch (err) {
      console.warn('Failed to save state:', err);
    }
  }, 1000); // Throttle saves to once per second
};

const preloadedState = loadState();

  reducer: {
    notes: notesReducer,
    ui: uiReducer,
    auth: authReducer,
```

### **Using confirmation hook:**

```javascript
import { useConfirm } from '../hooks/useConfirm';

const NoteActions = ({ noteId }) => {
  const { confirm, ConfirmDialog } = useConfirm();

  const handleDelete = async () => {
    const confirmed = await confirm({
      title: 'Delete note',
      message: 'This will move the note to trash.',
      confirmText: 'Delete',
      variant: 'danger',
    });

    if (confirmed) {
      deleteNote(noteId);
    }
  };

  return (
    <>
      <button onClick={handleDelete}>Delete</button>
      <ConfirmDialog />
    </>
  );
};
```
